Array.diff

using System;
public class Kata
{
  public static int[] ArrayDiff(int[] a, int[] b)
 {
     if (a.Length > 0)
        {
            int[] leftOvers = new int[a.Length];
            int counter = 0;
            for (int i = 0; i < a.Length; i++)
            {
                bool isokay = true;
                for (int j = 0; j < b.Length; j++)
                {
                    if (a[i] == b[j])
                    {
                        isokay = false;
                        break;
                    }
                }
                if (isokay)
                {
                    leftOvers[counter] = a[i];
                    counter++;
                }
            }
            Array.Resize(ref leftOvers, counter);
            
            return leftOvers;
        }
        else
            return a;
  }
}

using System.Text.RegularExpressions;
public static class Kata
{
  public static string AlphabetPosition(string text)
  {
    char[] allTextCharacters = text.ToCharArray();
    StringBuilder sb = new StringBuilder();
    Regex reg = new Regex("^[a-zA-Z]+$");
    foreach (char c in allTextCharacters)
      {
      if (reg.IsMatch(c.ToString()))
        {
      char upper = char.ToUpper(c);     
        int val = ((int)upper -(int)'A') + 1;
      sb.Append(val.ToString());
        sb.Append(" ");
    }
      }
    return sb.ToString().Trim();
  }
}

Convert string to camel case

using System;

public class Kata
{
  public static string ToCamelCase(string str)
  {
    string[]  inputStringArray = str.Split(new char[] {'-','_'}, StringSplitOptions.RemoveEmptyEntries);
    for(var i = 1; i<inputStringArray.Length; i++)
     inputStringArray[i] = string.Concat(char.ToUpper(inputStringArray[i][0]),inputStringArray[i].Substring(1)) ;
    str = string.Join("",inputStringArray);
    return str;
  }
}
